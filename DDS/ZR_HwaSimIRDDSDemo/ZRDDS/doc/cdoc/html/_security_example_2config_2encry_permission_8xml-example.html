SecurityExample/config/encry_permission.xml

该用例展示了内置安全插件数据加密的权限配置文件。

1 <?xml version="1.0" encoding="UTF-8"?>
2 <dds>
3  <permissions>
4  <grant name="app1">
5  <subject_name>C=CN,ST=JS,O=ZR,CN=app1</subject_name>
6  <validity>
7  <not_before>2020-10-19T10:54:00</not_before>
8  <not_after>2022-10-19T10:54:00</not_after>
9  </validity>
10  <default>ALLOW</default>
11  </grant>
12  <grant name="app2">
13  <subject_name>C=CN,ST=JS,O=ZR,CN=app2</subject_name>
14  <validity>
15  <not_before>2020-10-19T10:54:00</not_before>
16  <not_after>2022-10-19T10:54:00</not_after>
17  </validity>
18  <default>ALLOW</default>
19  </grant>
20  </permissions>
21 </dds>
