cmake_minimum_required(VERSION 3.5)
project(HwaSimIRCustomerReceiverDemo CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ZRDDS_ROOT)
  if(DEFINED ENV{ZRDDS_HOME})
    set(ZRDDS_ROOT "$ENV{ZRDDS_HOME}")
  else()
    message(FATAL_ERROR "Set ZRDDS_ROOT or ZRDDS_HOME")
  endif()
endif()
set(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
find_path(ZRDDS_CPP_INCLUDE ZRDDSCppSimpleInterface.h PATHS "${ZRDDS_ROOT}/include/CPlusPlusInterface" NO_DEFAULT_PATH)
find_library(ZRDDS_CPP_LIBRARY NAMES ZRDDSCpp_VS2015 ZRDDSCpp PATHS "${ZRDDS_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT ZRDDS_CPP_INCLUDE OR NOT ZRDDS_CPP_LIBRARY)
  message(FATAL_ERROR "ZRDDS C++ SDK not found under ${ZRDDS_ROOT}")
endif()
add_executable(HwaSimIRCustomerReceiverDemo main.cpp
  "${REPO_ROOT}/DDS/Generated/HwaSimIRProtocolV1/HwaSimIRProtocolV1.cpp"
  "${REPO_ROOT}/DDS/Generated/HwaSimIRProtocolV1/HwaSimIRProtocolV1TypeSupport.cpp")
target_compile_definitions(HwaSimIRCustomerReceiverDemo PRIVATE _ZRDDSCPPINTERFACE)
if(WIN32)
  target_compile_definitions(HwaSimIRCustomerReceiverDemo PRIVATE _ZRDDSDLLIMPORT)
endif()
target_include_directories(HwaSimIRCustomerReceiverDemo PRIVATE
  "${REPO_ROOT}/DDS/Generated/HwaSimIRProtocolV1"
  "${ZRDDS_ROOT}/include/CPlusPlusInterface" "${ZRDDS_ROOT}/include/ZRDDSCoreInterface")
target_link_libraries(HwaSimIRCustomerReceiverDemo PRIVATE "${ZRDDS_CPP_LIBRARY}")
if(UNIX)
  target_link_libraries(HwaSimIRCustomerReceiverDemo PRIVATE pthread dl)
  set_target_properties(HwaSimIRCustomerReceiverDemo PROPERTIES BUILD_RPATH "${ZRDDS_ROOT}/lib")
endif()
