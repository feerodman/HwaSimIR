417 template<typename T, typename TSeq>
418 ReturnCode_t ZRDDSDataReader<T, TSeq>::readOrTake(
419  TSeq &dataValues,
420  DataReaderReadOrTakeParams& params)
421 {
422  ReturnCode_t result = DDS_RETCODE_OK;
423  params.isLoan = false;
424  params.dataPtrArray = NULL;
425  params.dataCount = 0;
426  params.dataSeqLen = dataValues.length();
427  params.dataSeqMaxLen = dataValues.maximum();
428  params.dataSeq_has_ownership = dataValues.has_ownership();
429  params.dataSeqBuffer= (ZR_INT8*)dataValues.get_contiguous_buffer();
430  params.dataSize = sizeof(T);
431  params.dataSeqBufferType = 0;
432 
433  result = this->m_cppImpl->read_or_take_untype(params);
434  if (result == DDS_RETCODE_NO_DATA)
435  {
436  dataValues.length(0);
437  params.sampleInfos->length(0);
438  return result;
439  }
440 
441  if (result != DDS_RETCODE_OK)
442  {
443  return result;
444  }
445 
446  if (params.isLoan)
447  {
448  if (!dataValues.loan_discontiguous((T**)params.dataPtrArray, params.dataCount, params.dataCount))
449  {
450  result = DDS_RETCODE_ERROR;
451  this->m_cppImpl->return_loan_untype(params.dataPtrArray, params.dataCount, params.sampleInfos);
452  }
453  }
454  else
455  {
456  if (!dataValues.length(params.dataCount))
457  {
458  result = DDS_RETCODE_ERROR;
459  }
460  if (!params.sampleInfos->length(params.dataCount))
461  {
462  result = DDS_RETCODE_ERROR;
463  }
464  }
465  return result;
466 }

698 template<typename T, typename TSeq>
699 ReturnCode_t ZRDDSDataReader<T, TSeq>::return_loan(
700  TSeq &data_values,
701  SampleInfoSeq &sample_infos)
702 {
703  ReturnCode_t result = DDS_RETCODE_OK;
704  if (data_values.has_ownership() &&
705  sample_infos.has_ownership())
706  {
707  return result;
708  }
709 
710  Long dataSeqMaxLen = data_values.maximum();
711  T **dataSeqDiscontiguousBuffer = data_values.get_discontiguous_buffer();
712  result = this->m_cppImpl->return_loan_untype(
713  (void**)dataSeqDiscontiguousBuffer, dataSeqMaxLen, &sample_infos);
714  if (result != DDS_RETCODE_OK)
715  {
716  return result;
717  }
718  if (!data_values.unloan())
719  {
720  result = DDS_RETCODE_ERROR;
721  return result;
722  }
723  if (!sample_infos.unloan())
724  {
725  return DDS_RETCODE_ERROR;
726  }
727  return result;
728 }
