cmake_minimum_required(VERSION 3.18)
project(CustomerRawReceiverMinimal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ZRDDS_ROOT "" CACHE PATH "Installed ZRDDS SDK matching compiler and architecture")
find_path(ZRDDS_INCLUDE ZRDDSCppSimpleInterface.h PATHS "${ZRDDS_ROOT}/include/CPlusPlusInterface" NO_DEFAULT_PATH REQUIRED)
find_library(ZRDDS_LIBRARY NAMES ZRDDSCpp PATHS "${ZRDDS_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
add_executable(CustomerRawReceiverMinimal main.cpp ../CustomerTX/Generated/CustomerTxVideo.cpp ../CustomerTX/Generated/CustomerTxVideoTypeSupport.cpp)
target_compile_definitions(CustomerRawReceiverMinimal PRIVATE _ZRDDSCPPINTERFACE _ZRDDSDLLIMPORT)
target_include_directories(CustomerRawReceiverMinimal PRIVATE ../CustomerTX/Generated "${ZRDDS_INCLUDE}" "${ZRDDS_ROOT}/include/ZRDDSCoreInterface")
target_link_libraries(CustomerRawReceiverMinimal PRIVATE "${ZRDDS_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
