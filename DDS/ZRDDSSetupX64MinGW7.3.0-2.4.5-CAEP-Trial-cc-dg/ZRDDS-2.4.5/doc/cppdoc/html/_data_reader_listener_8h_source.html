184  virtual void on_data_available(DataReader *reader)
185  {
186  TSeq data_values;
187  SampleInfoSeq sample_infos;
188  /* 转化为类型安全接口 */
189  TDataReader* typedReader = (TDataReader*)reader;
190  ReturnCode_t retCode = typedReader->take(data_values, sample_infos,
191  LENGTH_UNLIMITED, ANY_SAMPLE_STATE, ANY_VIEW_STATE, ANY_INSTANCE_STATE);
192  if (RETCODE_OK != retCode)
193  {
194  printf("take failed(%d).\n", retCode);
195  return;
196  }
197  /* 处理数据 */
198  for (ZR_UINT32 i = 0; i < sample_infos._length; i++)
199  {
200  SampleInfo& info = sample_infos[i];
201  /* 判断是否为有效的负载数据 */
202  if (!info.valid_data)
203  {
204  continue;
205  }
206  T& sample = data_values[i];
207  this->on_process_sample(reader, sample, info);
208  }
209  typedReader->return_loan(data_values, sample_infos);
210  }
