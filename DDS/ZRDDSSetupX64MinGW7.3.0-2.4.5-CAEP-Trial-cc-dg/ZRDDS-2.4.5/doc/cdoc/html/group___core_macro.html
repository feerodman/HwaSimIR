核心接口常量

宏定义

#define DDS_XCDR_DATA_REPRESENTATION   0x0000
 网络传输的数据为使用CDR方式序列化后的样本。
 
#define DDS_XML_DATA_REPRESENTATION   0x0001
 网络传输的数据为使用XML方式方式序列化的样本。
 
#define DDS_ANY_INSTANCE_STATE   0xffffffff
 此掩码表示任意的实例状态 DDS_InstanceStateKind 。
 
#define DDS_NOT_ALIVE_INSTANCE_STATE   0x00000006
 此掩码表示非存活的实例状态 DDS_InstanceStateKind 。
 
#define MAX_USHORT_VALUE   0xffff
 2字节无符号整型最大值。
 
#define MAX_SHORT_VALUE   0x7fff
 2字节有符号整型最大值。
 
#define MAX_UINT32_VALUE   0xffffffff
 4字节无符号整型最大值。
 
#define MAX_INT32_VALUE   0x7fffffff
 4字节有符号整型最大值。
 
#define LENGTH_UNLIMITED   -1
 表明无限大，在不同的环境中取不同的值。
 
#define DDS_ANY_SAMPLE_STATE   0xffffffff
 此掩码表示任意的样本状态。
 
#define DDS_STATUS_MASK_ALL   0xffffffff
 此掩码表明所有类型的 DDS_StatusKind 的集合。
 
#define DDS_STATUS_MASK_NONE   0
 此掩码表明空的 DDS_StatusKind 的集合。
 
#define DDS_ANY_VIEW_STATE   0xffffffff
 此掩码表明所有类型的 DDS_ViewStateKind 的集合。
